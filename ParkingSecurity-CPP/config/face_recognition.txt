# DeepStream nvinfer config for the ArcFace face recognizer (buffalo_l/w600k_r50).
#
# Runs as a secondary classifier on every face the primary SCRFD detector
# produces; emits a raw 512-dim embedding tensor as user meta on each
# NvDsObjectMeta. The pad probe copies that into Detection.embedding; the
# embedding index does cosine matching against enrolled people elsewhere.
#
# ArcFace preprocessing (from insightface/arcface_onnx.py):
#   img = (pixel - 127.5) / 127.5   → pixels mapped to [-1, 1]
#   swapRB=True                     → model expects RGB
[property]
gpu-id=0

# 1/127.5 = 0.00784313725490196
net-scale-factor=0.00784313725490196
offsets=127.5;127.5;127.5
# 0 = RGB, 1 = BGR
model-color-format=0

model-engine-file=../models/w600k_r50.engine
labelfile-path=../config/face_labels.txt

batch-size=1
# network-mode: 0=FP32, 1=INT8, 2=FP16
network-mode=2
# process-mode: 1 = primary (whole frame), 2 = secondary (detected objects)
process-mode=2
# Must not collide with the detector's gie-unique-id=1.
gie-unique-id=2
# Consume faces from the detector (its gie-unique-id).
operate-on-gie-id=1
# ...class 0 (face).
operate-on-class-ids=0

# Input shape in explicit-batch mode.
infer-dims=3;112;112
force-implicit-batch-dim=0

# Skip crops smaller than this — ArcFace accuracy collapses below ~80px and
# those detections are the main source of "blurry-face" false positives.
input-object-min-width=80
input-object-min-height=80

# network-type: 0 = Detector, 1 = Classifier, 100 = Other (raw tensor only).
network-type=100
# Attach raw output tensors to NvDsObjectMeta user meta for our probe.
output-tensor-meta=1
classifier-async-mode=0
